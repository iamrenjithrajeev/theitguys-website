class Component extends DCLogic {
  constructor(props) {
    super(props);
    this.state = {
      floor: 'G', selected: 'B-106', panel: 'detail', emergency: false, ending: false, emStarted: '',
      overrides: {}, pinFor: '', pin: '', extra: 0,
      log: [
        { time: '10:42', unit: 'B-106', who: 'Customer M. Haddad', how: 'Bluetooth', result: 'Opened', kind: 'ok' },
        { time: '10:40', unit: 'Main gate', who: 'Rahul (shared, B-106)', how: 'Face terminal', result: 'Entered', kind: 'ok' },
        { time: '10:31', unit: 'C-104', who: 'Customer J. Thomas', how: 'NFC card', result: 'Opened', kind: 'ok' },

        { time: '10:12', unit: 'B-108', who: 'Customer S. Nair', how: 'One-time PIN', result: 'Opened', kind: 'ok' },
        { time: '09:00', unit: 'A-105', who: 'System', how: 'Auto-lockout', result: 'Access paused, payment failed', kind: 'warn' }
      ]
    };
  }
  now() { const d = new Date(); return String(d.getHours()).padStart(2, '0') + ':' + String(d.getMinutes()).padStart(2, '0'); }
  addLog(e) { this.setState({ log: [Object.assign({ time: this.now() }, e)].concat(this.state.log) }); }
  baseStatus(id, k) {
    if (id === 'B-108') return 'open';
    if (k % 13 === 4) return 'overdue';
    if (k % 19 === 7) return 'alert';
    if (k % 6 === 2) return 'vacant';
    return 'occupied';
  }
  renderVals() {
    const s = this.state;
    const names = ['M. Haddad', 'S. Nair', 'J. Thomas', 'F. Al Ali', 'R. Menon', 'L. Costa', 'A. Khan', 'D. Pereira'];
    const plans = ['Smart', 'Smart', 'Premium', 'Basic'];
    const look = {
      occupied: ['#E8EEF7', '1px solid #9FB3D1', '#1E3A5F'],
      vacant: ['#FFFFFF', '1px dashed #9AA5B4', '#556070'],
      overdue: ['#FDF1E3', '1px solid #D9822B', '#8A4B0F'],
      open: ['#E3F4EE', '1px solid #0E7C66', '#0B5E4D'],
      alert: ['#FBE9E9', '1px solid #C0392B', '#8E1F1F']
    };
    const words = { occupied: 'Rented, locked', vacant: 'Vacant, locked', overdue: 'Access paused, unpaid', open: 'Open now', alert: 'Low battery, check lock' };
    const layout = s.floor === 'G'
      ? [['A', 16], ['B', 16], ['C', 16], ['D', 16]]
      : [['E', 8 + s.extra], ['F', 10]];
    const info = {};
    let k = s.floor === 'G' ? 0 : 64;
    const counts = { total: 0, rented: 0, vacant: 0, overdue: 0, alert: 0, open: 0 };
    const corridors = layout.map((row) => {
      const c = row[0], n = row[1];
      const units = [];
      for (let i = 0; i < n; i++) {
        const id = c + '-' + (s.floor === 'G' ? 101 + i : 201 + i);
        const status = s.overrides[id] || this.baseStatus(id, k);
        const sel = id === s.selected;
        const L = look[status];
        info[id] = {
          id: id, status: status, statusText: words[status],
          tenant: status === 'vacant' ? 'No customer' : names[k % names.length],
          plan: status === 'vacant' ? 'None' : plans[k % plans.length],
          battery: (status === 'alert' ? 14 : 55 + (k * 7) % 45) + '%',
          gateway: 'G3-' + c + (i < n / 2 ? '1' : '2'),
          pill: 'font-size: 13px; font-weight: 600; padding: 6px 10px; border-radius: 999px; background: ' + L[0] + '; color: ' + L[2] + '; border: ' + L[1]
        };
        counts.total++;
        if (status !== 'vacant') counts.rented++;
        if (status === 'vacant') counts.vacant++;
        if (status === 'overdue') counts.overdue++;
        if (status === 'alert') counts.alert++;
        if (status === 'open') counts.open++;
        units.push({
          short: id,
          aria: 'Unit ' + id + ', ' + words[status],
          style: 'min-height: 44px; border-radius: 8px; cursor: pointer; font-size: 12px; font-weight: 600; background: ' + L[0] + '; border: ' + L[1] + '; color: ' + L[2] + '; outline: ' + (sel ? '3px solid #12233D' : 'none') + '; outline-offset: 1px; opacity: ' + (s.emergency && status !== 'open' && !sel ? '0.55' : '1'),
          pick: () => this.setState({ selected: id, panel: 'detail' })
        });
        k++;
      }
      return { name: 'Corridor ' + c, gateways: 'Gateways G3-' + c + '1, G3-' + c + '2 online', units: units };
    });
    const sel = info[s.selected] || info[corridors[0].units[0].short];
    const tab = (on) => 'min-height: 44px; padding: 0 14px; border-radius: 10px; cursor: pointer; font-size: 14px; font-weight: 600; border: 1px solid ' + (on ? '#12233D' : '#D4D8DE') + '; background: ' + (on ? '#12233D' : '#FFFFFF') + '; color: ' + (on ? '#FFFFFF' : '#0F172A');
    const pills = { ok: ['#E3F4EE', '#0B5E4D'], bad: ['#FBE9E9', '#8E1F1F'], warn: ['#FDF1E3', '#8A4B0F'], info: ['#E8EEF7', '#1E3A5F'] };
    const approve = (reason) => () => {
      this.setState({ panel: 'approving' });
      setTimeout(() => {
        const o = Object.assign({}, this.state.overrides); o[sel.id] = 'open';
        this.setState({ panel: 'approved', overrides: o });
        this.addLog({ unit: sel.id, who: 'Manager, approved by Owner', how: 'Remote override: ' + reason, result: 'Opened, clip saved', kind: 'warn' });
      }, 1600);
    };
    return {
      emergency: s.emergency,
      emStarted: s.emStarted,
      emButton: s.emergency
        ? 'min-height: 44px; padding: 0 16px; border-radius: 10px; border: none; background: #B91C1C; color: #FFFFFF; font-size: 14px; font-weight: 700; cursor: pointer'
        : 'min-height: 44px; padding: 0 16px; border-radius: 10px; border: 2px solid #F87171; background: transparent; color: #FFFFFF; font-size: 14px; font-weight: 700; cursor: pointer',
      emButtonLabel: s.emergency ? 'Emergency active' : 'Emergency',
      toggleEmergency: () => {
        if (s.emergency) return;
        this.setState({ emergency: true, emStarted: this.now() });
        this.addLog({ unit: 'Whole site', who: 'Fire alarm panel', how: 'Fire relay', result: 'Emergency mode on, gates released', kind: 'bad' });
      },
      endLabel: s.ending ? 'Waiting for second approval…' : 'End emergency (needs a second approval)',
      endEmergency: () => {
        if (s.ending) return;
        this.setState({ ending: true });
        setTimeout(() => {
          this.setState({ emergency: false, ending: false });
          this.addLog({ unit: 'Whole site', who: 'Manager, approved by Owner', how: 'Admin panel', result: 'Emergency ended, gates normal', kind: 'info' });
        }, 1600);
      },
      stats: counts,
      isFirst: s.floor === '1',
      goGround: () => this.setState({ floor: 'G', selected: 'B-106', panel: 'detail' }),
      goFirst: () => this.setState({ floor: '1', selected: 'E-201', panel: 'detail' }),
      tabGround: tab(s.floor === 'G'), tabFirst: tab(s.floor === '1'),
      addUnit: () => { this.setState({ extra: s.extra + 1 }); this.addLog({ unit: 'E-' + (209 + s.extra), who: 'Manager', how: 'Unit editor', result: 'New unit created, padlock paired', kind: 'info' }); },
      corridors: corridors,
      sel: sel,
      panelDetail: s.panel === 'detail', panelReason: s.panel === 'reason', panelApproving: s.panel === 'approving', panelApproved: s.panel === 'approved',
      startUnlock: () => this.setState({ panel: 'reason' }),
      cancelUnlock: () => this.setState({ panel: 'detail' }),
      backToDetail: () => this.setState({ panel: 'detail' }),
      reasonLost: approve('customer lost phone'),
      reasonMaint: approve('maintenance'),
      reasonPolice: approve('police request'),
      hasPin: s.pinFor === sel.id,
      pinShown: s.pin,
      issuePin: () => {
        const p = String(Math.floor(100000 + Math.random() * 900000));
        this.setState({ pinFor: sel.id, pin: p.slice(0, 3) + ' ' + p.slice(3) });
        this.addLog({ unit: sel.id, who: 'Manager', how: 'Admin panel', result: 'One-time PIN issued, 30 min', kind: 'info' });
      },
      pauseLabel: sel.status === 'overdue' ? 'Restore access' : 'Pause access',
      togglePause: () => {
        const o = Object.assign({}, s.overrides);
        const paused = sel.status === 'overdue';
        o[sel.id] = paused ? 'occupied' : 'overdue';
        this.setState({ overrides: o });
        this.addLog({ unit: sel.id, who: 'Manager', how: 'Admin panel', result: paused ? 'Access restored' : 'Access paused', kind: paused ? 'ok' : 'warn' });
      },
      log: s.log.slice(0, 10).map((l) => Object.assign({}, l, { pill: 'font-size: 13px; font-weight: 600; padding: 4px 10px; border-radius: 999px; white-space: normal; background: ' + pills[l.kind][0] + '; color: ' + pills[l.kind][1] }))
    };
  }
